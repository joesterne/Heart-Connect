   157	@OptIn(ExperimentalMaterial3Api::class)
   158	@Composable
   159	fun CommunityFeed(
   160	    posts: List<CommunityPost>,
   161	    savedPosts: Set<String>,
   162	    onToggleSave: (String) -> Unit,
   163	    onCreatePost: (String) -> Unit
   164	) {
   165	    var newPostText by remember { mutableStateOf("") }
   166	    var selectedFilter by remember { mutableStateOf("All Phases") }
   167	    var filterExpanded by remember { mutableStateOf(false) }
   168	    var searchQuery by remember { mutableStateOf("") }
   169	    
   170	    val journeyPhases = listOf("All Phases", "Pre-transplant", "Waitlisted", "Post-transplant recovery", "Long-term survivor", "Caregiver")
   171	    val filteredPosts = remember(posts, selectedFilter, searchQuery) {
   172	        posts.filter { post ->
   173	            val phaseMatch = selectedFilter == "All Phases" || post.authorJourneyPhase == selectedFilter
   174	            val searchMatch = searchQuery.isBlank() || 
   175	                post.content.contains(searchQuery, ignoreCase = true) || 
   176	                post.authorName.contains(searchQuery, ignoreCase = true)
   177	            phaseMatch && searchMatch
   178	        }
   179	    }
   180	    
   181	    Column(modifier = Modifier.fillMaxSize()) {
   182	        Card(
   183	            modifier = Modifier.fillMaxWidth().padding(16.dp),
   184	            shape = RoundedCornerShape(24.dp),
   185	            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surfaceVariant)
   186	        ) {
   187	            Column(modifier = Modifier.padding(16.dp)) {
   188	                OutlinedTextField(
   189	                    value = searchQuery,
   190	                    onValueChange = { searchQuery = it },
   191	                    placeholder = { Text("Search posts or members...") },
   192	                    leadingIcon = { Icon(androidx.compose.material.icons.Icons.Default.Search, contentDescription = "Search") },
   193	                    modifier = Modifier.fillMaxWidth(),
   194	                    shape = RoundedCornerShape(24.dp),
   195	                    singleLine = true
   196	                )
   197	                Spacer(modifier = Modifier.height(16.dp))
   198	                OutlinedTextField(
   199	                    value = newPostText,
   200	                    onValueChange = { newPostText = it },
   201	                    placeholder = { Text("Share your experience or ask a question...") },
   202	                    modifier = Modifier.fillMaxWidth(),
   203	                    shape = RoundedCornerShape(24.dp),
   204	                    minLines = 3
   205	                )
   206	                Spacer(modifier = Modifier.height(8.dp))
   207	                Row(
   208	                    modifier = Modifier.fillMaxWidth(),
   209	                    horizontalArrangement = Arrangement.SpaceBetween,
   210	                    verticalAlignment = Alignment.CenterVertically
   211	                ) {
   212	                    ExposedDropdownMenuBox(
   213	                        expanded = filterExpanded,
   214	                        onExpandedChange = { filterExpanded = !filterExpanded },
   215	                        modifier = Modifier.weight(1f).padding(end = 8.dp)
   216	                    ) {
   217	                        OutlinedTextField(
   218	                            value = selectedFilter,
   219	                            onValueChange = {},
   220	                            readOnly = true,
   221	                            label = { Text("Filter by Phase") },
   222	                            trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expanded = filterExpanded) },
   223	                            modifier = Modifier.fillMaxWidth().menuAnchor(MenuAnchorType.PrimaryNotEditable),
   224	                            shape = RoundedCornerShape(24.dp)
   225	                        )
   226	                        ExposedDropdownMenu(
   227	                            expanded = filterExpanded,
   228	                            onDismissRequest = { filterExpanded = false }
   229	                        ) {
   230	                            journeyPhases.forEach { phase ->
   231	                                DropdownMenuItem(
   232	                                    text = { Text(phase) },
   233	                                    onClick = {
   234	                                        selectedFilter = phase
   235	                                        filterExpanded = false
   236	                                    }
   237	                                )
   238	                            }
   239	                        }
   240	                    }
   241	                    val context = androidx.compose.ui.platform.LocalContext.current
   242	                    Button(
   243	                        onClick = {
   244	                            if (newPostText.trim().length < 5) {
   245	                                android.widget.Toast.makeText(context, "Post must be at least 5 characters.", android.widget.Toast.LENGTH_SHORT).show()
   246	                            } else {
   247	                                onCreatePost(newPostText.trim())
   248	                                newPostText = ""
   249	                            }
   250	                        },
   251	                        shape = RoundedCornerShape(24.dp),
   252	                        modifier = Modifier.height(56.dp)
   253	                    ) {
   254	                        Text("Post")
   255	                    }
   256	                }
   257	            }
   258	        }
   259	        
   260	        LazyColumn(
   261	            modifier = Modifier.fillMaxSize(),
   262	            contentPadding = PaddingValues(horizontal = 16.dp, vertical = 8.dp),
   263	            verticalArrangement = Arrangement.spacedBy(12.dp)
   264	        ) {
   265	            items(items = filteredPosts, key = { it.id }) { post ->
   266	                PostCard(
   267	                    post = post,
   268	                    isSaved = savedPosts.contains(post.id),
   269	                    onToggleSave = { onToggleSave(post.id) }
   270	                )
   271	            }
   272	        }
   273	    }
   274	}

   522	@Composable
   523	fun QACard(
   524	    inquiry: com.example.data.model.QAInquiry,
   525	    isMentor: Boolean,
   526	    onAnswer: (String) -> Unit,
   527	    onVerify: (String) -> Unit
   528	) {
   529	    var showAnswerField by remember { mutableStateOf(false) }
   530	    var answerText by remember { mutableStateOf("") }
   531	
   532	    Card(
   533	        modifier = Modifier.fillMaxWidth(),
   534	        shape = RoundedCornerShape(24.dp),
   535	        colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface)
   536	    ) {
   537	        Column(modifier = Modifier.padding(16.dp)) {
   538	            Text(inquiry.question, style = MaterialTheme.typography.titleMedium, fontWeight = FontWeight.Bold)
   539	            Spacer(modifier = Modifier.height(4.dp))
   540	            Text(
   541	                "Asked by ${inquiry.authorName} • ${SimpleDateFormat("MMM dd", Locale.getDefault()).format(Date(inquiry.timestamp))}",
   542	                style = MaterialTheme.typography.labelSmall,
   543	                color = MaterialTheme.colorScheme.onSurfaceVariant
   544	            )
   545	            Spacer(modifier = Modifier.height(16.dp))
   546	            
   547	            inquiry.answers.forEach { answer ->
   548	                Surface(
   549	                    modifier = Modifier.fillMaxWidth().padding(bottom = 8.dp),
   550	                    color = if (answer.isVerified) MaterialTheme.colorScheme.primaryContainer else MaterialTheme.colorScheme.surfaceVariant,
   551	                    shape = RoundedCornerShape(16.dp)
   552	                ) {
   553	                    Column(modifier = Modifier.padding(12.dp)) {
   554	                        Row(verticalAlignment = Alignment.CenterVertically) {
   555	                            Text(answer.authorName, style = MaterialTheme.typography.labelMedium, fontWeight = FontWeight.Bold)
   556	                            if (answer.isVerified) {
   557	                                Spacer(modifier = Modifier.width(4.dp))
   558	                                Icon(Icons.Default.Verified, contentDescription = "Verified", tint = MaterialTheme.colorScheme.primary, modifier = Modifier.size(14.dp))
   559	                                Text(" Verified", style = MaterialTheme.typography.labelSmall, color = MaterialTheme.colorScheme.primary)
   560	                            }
   561	                        }
   562	                        Spacer(modifier = Modifier.height(4.dp))
   563	                        Text(answer.content, style = MaterialTheme.typography.bodyMedium)
   564	                        
   565	                        if (isMentor && !answer.isVerified) {
   566	                            Spacer(modifier = Modifier.height(8.dp))
   567	                            TextButton(onClick = { onVerify(answer.id) }) {

   570	                        }

   572	                }
   573	            }
   574	
   575	            if (showAnswerField) {

   585	                        Text("Cancel")
   586	                    }
   587	                    Button(onClick = { 
   588	                        if (answerText.isNotBlank()) {
   589	                            onAnswer(answerText)
   590	                            showAnswerField = false
   591	                            answerText = ""
   592	                        }
   593	                    }) {
   594	                        Text("Submit")
   595	                    }
   596	                }
   597	            } else {
   598	                TextButton(onClick = { showAnswerField = true }, modifier = Modifier.padding(top = 8.dp)) {
   599	                    Text("Add an Answer")
   600	                }
   601	            }
   602	        }
   603	            }
   604	        }
